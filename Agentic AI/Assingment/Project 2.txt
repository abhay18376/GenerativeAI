"""
langgraph_customer_support_agent.py

Self-contained example demonstrating an AI customer-support agent using a
conceptual LangGraph-like state and memory management pattern.

Features implemented (maps to user's steps):
- AgentState dataclass with short-term and long-term memory + metadata (step 1)
- StateGraph + reducer function to manage state transitions (step 2)
- Short-term memory (session messages) and long-term memory (persistent user_history)
  with a MemorySaver abstraction that supports an in-memory and JSON file backend (step 3)
- Nodes: MockLLM node for answering common queries, HistoryTool node to fetch user history
  and personalization (step 4)
- Message trimming and filtering to keep only recent actionable messages (step 5)
- Simple HITL escalation for "complex" queries: escalates to a human queue

Note: This code uses only standard Python libraries and implements minimal
abstractions so it can run as-is (no external LangGraph dependency required).

Run as a script to see a short demo.
"""
from __future__ import annotations
import json
from dataclasses import dataclass, field, asdict
from typing import List, Dict, Any, Callable, Optional
import re
import uuid
import datetime

# ----------------------------- Step 1: AgentState -----------------------------
@dataclass
class Message:
    sender: str  # 'user' | 'agent' | 'system' | 'human'
    text: str
    timestamp: str = field(default_factory=lambda: datetime.datetime.utcnow().isoformat())

@dataclass
class AgentState:
    user_id: str
    thread_id: str
    messages: List[Message] = field(default_factory=list)  # short-term memory for current session
    user_history: List[Dict[str, Any]] = field(default_factory=list)  # long-term memory
    metadata: Dict[str, Any] = field(default_factory=dict)

    def append_message(self, sender: str, text: str):
        self.messages.append(Message(sender=sender, text=text))

# ----------------------------- MemorySaver abstraction -----------------------------
class MemorySaver:
    """Abstract memory saver. Implementations should provide `save_user_history` and `load_user_history`."""
    def save_user_history(self, user_id: str, user_history: List[Dict[str, Any]]):
        raise NotImplementedError

    def load_user_history(self, user_id: str) -> List[Dict[str, Any]]:
        raise NotImplementedError

class InMemoryMemorySaver(MemorySaver):
    def __init__(self):
        self.store: Dict[str, List[Dict[str, Any]]] = {}

    def save_user_history(self, user_id: str, user_history: List[Dict[str, Any]]):
        self.store[user_id] = user_history.copy()

    def load_user_history(self, user_id: str) -> List[Dict[str, Any]]:
        return self.store.get(user_id, []).copy()

class JSONFileMemorySaver(MemorySaver):
    def __init__(self, path: str = "user_histories.json"):
        self.path = path
        try:
            with open(self.path, "r", encoding="utf-8") as f:
                self.store = json.load(f)
        except FileNotFoundError:
            self.store = {}

    def _persist(self):
        with open(self.path, "w", encoding="utf-8") as f:
            json.dump(self.store, f, indent=2)

    def save_user_history(self, user_id: str, user_history: List[Dict[str, Any]]):
        self.store[user_id] = user_history
        self._persist()

    def load_user_history(self, user_id: str) -> List[Dict[str, Any]]:
        return self.store.get(user_id, [])

# ----------------------------- Step 2: StateGraph & Reducer -----------------------------
class StateGraph:
    """Minimal state manager that applies reducers to the AgentState."""
    def __init__(self, initial_state: AgentState, reducer: Callable[[AgentState, Dict[str, Any]], AgentState]):
        self.state = initial_state
        self.reducer = reducer

    def dispatch(self, action: Dict[str, Any]):
        # action is a dict like {"type": "append_message", "payload": {...}}
        self.state = self.reducer(self.state, action)
        return self.state

# Reducer example: handles appending messages and updating user_history
def agent_state_reducer(state: AgentState, action: Dict[str, Any]) -> AgentState:
    t = action.get("type")
    p = action.get("payload", {})

    if t == "append_message":
        state.append_message(sender=p["sender"], text=p["text"])
        return state

    if t == "replace_messages":
        state.messages = p.get("messages", [])
        return state

    if t == "add_user_history":
        entry = p.get("entry")
        if entry:
            state.user_history.append(entry)
        return state

    # other action types could be handled here
    return state

# ----------------------------- Step 4: Nodes (Mock LLM, Tools) -----------------------------
class MockLLM:
    """A tiny rule-based LLM substitute for demo/testing.
    In production, swap this with a real LLM node.
    """
    COMMON_QA = {
        "how do i reset my password": "To reset your password, go to Settings -> Security -> Reset Password. If you can't access your account, use 'Forgot password' on the sign-in page.",
        "what is my billing balance": "Your current billing balance is shown in the Billing section of your account dashboard.",
    }

    def __init__(self):
        pass

    def respond(self, prompt: str, context: Optional[AgentState] = None) -> str:
        clean = prompt.strip().lower()
        # naive exact match first
        for q, a in self.COMMON_QA.items():
            if q in clean:
                return a

        # personalize if we have history
        if context and context.user_history:
            last = context.user_history[-1]
            return f"I see from your history that you previously asked about '{last.get('topic')}'. How can I help further?"

        # fallback heuristic
        return "I'm not sure about that. I'll escalate this to a human agent to help you further."

class HistoryTool:
    """Fetches user history from MemorySaver and returns a summary."""
    def __init__(self, memory_saver: MemorySaver):
        self.ms = memory_saver

    def fetch_summary(self, user_id: str, limit: int = 3) -> str:
        hist = self.ms.load_user_history(user_id)
        if not hist:
            return "No previous history found."
        entries = hist[-limit:]
        lines = [f"{e.get('timestamp','?')}: {e.get('topic','(no topic)')}" for e in entries]
        return "Last interactions:\n" + "\n".join(lines)

# ----------------------------- Step 3: Short-term & Long-term Memory usage -----------------------------
class CustomerSupportAgent:
    def __init__(self, memory_saver: MemorySaver, llm: MockLLM, hitl_queue: Optional[List[Dict]] = None):
        self.memory_saver = memory_saver
        self.llm = llm
        self.hitl_queue = hitl_queue if hitl_queue is not None else []

    def start_session(self, user_id: str, thread_id: Optional[str] = None) -> StateGraph:
        thread_id = thread_id or str(uuid.uuid4())
        state = AgentState(user_id=user_id, thread_id=thread_id)
        # load long-term history into state
        state.user_history = self.memory_saver.load_user_history(user_id)
        sg = StateGraph(initial_state=state, reducer=agent_state_reducer)
        return sg

    # message filter: remove useless greetings and short non-actionable messages
    def _filter_message(self, text: str) -> bool:
        text_clean = text.strip().lower()
        if len(text_clean) < 2:
            return False
        if re.fullmatch(r"(hi|hello|hey|thanks|thank you|tnx)[.!]*", text_clean):
            return False
        return True

    # trimming: keep only last n messages
    def _trim_messages(self, state: AgentState, keep: int = 5):
        if len(state.messages) <= keep:
            return
        state.messages = state.messages[-keep:]

    def _is_complex(self, text: str) -> bool:
        # simple complexity heuristic: long messages or certain keywords
        keywords = ["chargeback", "legal", "complaint", "refund not received", "escalate", "data breach"]
        if any(k in text.lower() for k in keywords):
            return True
        if len(text.split()) > 40:  # very long message
            return True
        # if LLM returns fallback text, we'll escalate elsewhere
        return False

    def handle_user_message(self, sg: StateGraph, text: str) -> Dict[str, Any]:
        state = sg.state
        # step: filter
        if self._filter_message(text):
            sg.dispatch({"type": "append_message", "payload": {"sender": "user", "text": text}})
        else:
            # filtered out: treat as system message, but still append minimal record
            sg.dispatch({"type": "append_message", "payload": {"sender": "system", "text": f"filtered: {text}"}})

        # trimming
        self._trim_messages(sg.state, keep=5)

        # HITL decision
        if self._is_complex(text):
            # prepare context snapshot and push to human queue
            ticket = {
                "ticket_id": str(uuid.uuid4()),
                "user_id": state.user_id,
                "thread_id": state.thread_id,
                "messages": [asdict(m) for m in state.messages],
                "reason": "complex_heuristic",
                "timestamp": datetime.datetime.utcnow().isoformat(),
            }
            self.hitl_queue.append(ticket)
            # record in long-term memory that this ticket was created
            sg.dispatch({"type": "add_user_history", "payload": {"entry": {"timestamp": ticket["timestamp"], "topic": "Escalated to human", "ticket_id": ticket["ticket_id"]}}})
            self.memory_saver.save_user_history(state.user_id, state.user_history)
            return {"reply": "I've escalated this to a human agent. They'll review and get back to you.", "escalated": True}

        # Use history tool to personalize
        history_tool = HistoryTool(self.memory_saver)
        hist_summary = history_tool.fetch_summary(state.user_id)

        # Construct LLM prompt (simple)
        prompt = f"History:\n{hist_summary}\n\nRecent messages:\n" + "\n".join(m.text for m in state.messages[-5:]) + f"\n\nUser asked: {text}"
        llm_resp = self.llm.respond(prompt, context=state)

        # If LLM fallback indicates uncertainty, escalate to HITL as well
        if "escalate" in llm_resp.lower() or "not sure" in llm_resp.lower():
            ticket = {
                "ticket_id": str(uuid.uuid4()),
                "user_id": state.user_id,
                "thread_id": state.thread_id,
                "messages": [asdict(m) for m in state.messages],
                "reason": "llm_uncertain",
                "timestamp": datetime.datetime.utcnow().isoformat(),
            }
            self.hitl_queue.append(ticket)
            sg.dispatch({"type": "add_user_history", "payload": {"entry": {"timestamp": ticket["timestamp"], "topic": "LLM Escalation", "ticket_id": ticket["ticket_id"]}}})
            self.memory_saver.save_user_history(state.user_id, state.user_history)
            return {"reply": "I wasn't able to fully resolve that. I've escalated this to a human agent.", "escalated": True}

        # Append agent reply to short-term memory and optionally long-term summary
        sg.dispatch({"type": "append_message", "payload": {"sender": "agent", "text": llm_resp}})
        # Optional: record resolved queries as history entries
        sg.dispatch({"type": "add_user_history", "payload": {"entry": {"timestamp": datetime.datetime.utcnow().isoformat(), "topic": text[:80], "resolution": llm_resp}}})
        self.memory_saver.save_user_history(state.user_id, state.user_history)

        return {"reply": llm_resp, "escalated": False}

# ----------------------------- Demo / Example usage -----------------------------
def demo():
    print("Starting demo of customer support agent...\n")
    memory = InMemoryMemorySaver()
    llm = MockLLM()
    agent = CustomerSupportAgent(memory_saver=memory, llm=llm)

    # start a session
    sg = agent.start_session(user_id="user_123")

    # simulate conversation
    queries = [
        "Hi",
        "How do I reset my password?",
        "I also have a question about my last charge — I was billed twice for order 45678",
        "Also, I want a refund for that order, it's been 40 days and no refund received.",
    ]

    for q in queries:
        print(f"USER: {q}")
        out = agent.handle_user_message(sg, q)
        print("AGENT:", out["reply"])
        if out.get("escalated"):
            print("-> This message was escalated to human support.")
        print("---")

    # show final state
    print("Final short-term messages:")
    for m in sg.state.messages:
        print(f" - {m.sender}: {m.text}")

    print("\nLong-term history stored:")
    for h in sg.state.user_history:
        print(f" - {h}")

    print("\nHITL queue contents:")
    for t in agent.hitl_queue:
        print(f" - ticket {t['ticket_id']} reason: {t['reason']}")

if __name__ == '__main__':
    demo()
